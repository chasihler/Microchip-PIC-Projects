/** \brief Sends a stop signal to the I2C bus.  
  
  This function is meant to be used by higher level  
  funtions to communicate with a I2C device.  
  \sa i2c_wait_for_scl(), i2c_read_acknowledge(),  
  i2c_send_byte(), i2c_read_byte(), i2c_open().  
  \ingroup i2c_software  
*/   
void i2c_stop( void ) {   
    I2C_SCL_LOW();          // assume SCL is high on entry   
    I2C_SDA_LOW();          // ensure the data is low first   
    I2C_DELAY_1uS();   
    I2C_SCL_DIR = I2C_INPUT;   
    delay_10us();   
    I2C_SDA_HIGH();         // the low->high data transistion   
    delay_10us();           // bus free time before next start   
}   
   
/** \brief Waits for the clock to be released by the slave device.  
  
  This function is meant to be used by higher level  
  funtions to communicate with a I2C device.  
  \sa i2c_stop(), i2c_read_acknowledge(),  
  i2c_send_byte(), i2c_read_byte(), i2c_open().  
  \return TRUE if the wait failed, FALSE otherwise.  
  \ingroup i2c_software  
*/   
UBYTE i2c_wait_for_scl( void ) {   
    if ( !I2C_SCL ) {       // SCL_DIR should be input here   
        delay_10us();   
        if ( !I2C_SCL ) {   // if the clock is still low -> bus error   
            return TRUE;   
        }   
    }   
    return FALSE;   
}   
   
/** \brief Reads acknowledge signal from slave device.  
  
  This function is meant to be used by higher level  
  funtions to communicate with a I2C device.  
  \return I2C_ERROR if bus error or,  
  if slave acknowledges it returns ack = 0, ack = 1 otherwise.  
  \sa i2c_stop(), i2c_wait_for_scl(),  
  i2c_send_byte(), i2c_read_byte(), i2c_open().  
  \ingroup i2c_software  
*/   
UBYTE i2c_read_acknowledge( void ) {   
    UBYTE ack;   
   
    I2C_SCL_LOW();   
    I2C_SDA_DIR = I2C_INPUT;// disable data line - listen for ack   
    delay_10us();               // SCL low to data out valid   
    I2C_SCL_DIR = I2C_INPUT;// float clock high   
    delay_10us();   
    ack = I2C_SDA;          // read the acknowledge   
    if ( i2c_wait_for_scl() )   
        return I2C_ERROR;   
    return ( ack );   
}   
   
/** \brief Sends a byte through the I2C bus.  
  
  This function is meant to be used by higher level  
  funtions to communicate with a I2C device.  
  \param byte_to_send Specifies the byte to send.  
  \return I2C_ERROR if bus error or,  
  it returns the output of i2c_read_acknowledge().  
  \sa i2c_stop(), i2c_read_acknowledge(), i2c_wait_for_scl(),  
  i2c_read_acknowledge(), i2c_read_byte(), i2c_open().  
  \ingroup i2c_software  
*/   
UBYTE i2c_send_byte( UBYTE byte_to_send ) {   
    SBYTE i;   
   
    for ( i=7; i>=0; i-- ) {   
        I2C_SCL_LOW();                          // ensure clock is low   
        I2C_SDA_DIR = ((byte_to_send>>i)&0x01);   // bit to send   
        I2C_DELAY_1uS();   
        I2C_SCL_DIR = I2C_INPUT;                // float clock high   
        if ( i2c_wait_for_scl() )   
            return I2C_ERROR;   
        delay_10us();                               // clock high time   
    }   
    return ( i2c_read_acknowledge() );          // returns ack, ~ack   
}   
   
/** \brief Reads a byte on the I2C bus.  
  
  This function is meant to be used by higher level  
  funtions to communicate with a I2C device.  
  \return I2C_ERROR if bus error or,  
  it returns the byte read.  
  \sa i2c_stop(), i2c_read_acknowledge(), i2c_wait_for_scl(),  
  i2c_read_acknowledge(), i2c_send_byte(), i2c_open().  
  \ingroup i2c_software  
*/   
UBYTE i2c_read_byte( void ) {   
    SBYTE i;   
    UBYTE _byte;   
       
    _byte = 0;   
    for ( i=7;i>=0;i-- ) {   
        I2C_SCL_LOW();              // ensure clock is low   
        delay_10us();                   // min clock low  period   
        I2C_SDA_DIR = I2C_INPUT;    // release data line   
        I2C_SCL_DIR = I2C_INPUT;    // float clock high   
        if ( i2c_wait_for_scl() )   
            return I2C_ERROR;   
        delay_10us();   
        _byte    = _byte << 1;        // read the next bit   
        _byte   |= I2C_SDA;   
    }   
    return _byte;   
}             
   
/** \brief Opens and send the control byte to the I2C bus.  
  
  This function is meant to be used by higher level  
  funtions to communicate with a I2C device.  
  \param control_byte Specifies the device's control byte.  
  \return The output of i2c_send_byte().  
  \sa i2c_stop(), i2c_read_acknowledge(), i2c_wait_for_scl(),  
  i2c_read_acknowledge(), i2c_send_byte(), i2c_read_byte().  
  \ingroup i2c_software  
*/   
UBYTE i2c_open( UBYTE control_byte ) {   
    I2C_SCL_LOW();                  // ensure clock is low   
    I2C_SDA_HIGH();                 // ensure data is high   
    I2C_DELAY_1uS();   
    I2C_SCL_DIR = I2C_INPUT;        // clock pulse high   
    delay_10us();   
    I2C_SDA_LOW();                  // the high->low transition   
    delay_10us();   
    return ( i2c_send_byte(control_byte) );   
}

/** \brief Writes data to an I2C EEPROM device.  
  
  This function uses the I2C support functions to communicate  
  with an I2C EEPROM made by Microchip (tm).  
  \sa i2c_stop(), i2c_read_acknowledge(), i2c_wait_for_scl(),  
  i2c_read_acknowledge(), i2c_send_byte(), i2c_read_byte().  
  \ingroup i2c_software  
*/   
void EEPROMPageWrite( UBYTE cbyte, UBYTE addr, UBYTE size, bank1 UBYTE* ram ) {   
    CLRWDT();   
    INTERRUPT_GLOBAL_DISABLE();   
    while ( i2c_open(cbyte) );   
    i2c_send_byte( addr );   
    while ( size-- ) {   
        i2c_send_byte(*ram);   
        ram++;   
    }   
    i2c_stop();   
    INTERRUPT_GLOBAL_ENABLE();   
}   
   
/** \brief Reads data to an I2C EEPROM device.  
  
  This function uses the I2C support functions to communicate  
  with an I2C EEPROM made by Microchip (tm).  
  \sa i2c_stop(), i2c_read_acknowledge(), i2c_wait_for_scl(),  
  i2c_read_acknowledge(), i2c_send_byte(), i2c_read_byte().  
  \ingroup i2c_software  
*/   
void EEPROMSequentialRead( UBYTE cbyte, UBYTE addr, UBYTE size, bank1 UBYTE* ram ) {   
    CLRWDT();   
    INTERRUPT_GLOBAL_DISABLE();   
    while ( i2c_open(cbyte) );   
    i2c_send_byte( addr );   
    cbyte |= 1;   
    while ( i2c_open(cbyte) );   
    while ( size-- ) {   
        *ram = i2c_read_byte();   
        ram++;   
        I2C_SCL_LOW();   
        I2C_SDA_DIR = size?LOW:HIGH;    // drive line low -> more to come   
        I2C_DELAY_1uS();   
        I2C_SCL_DIR = I2C_INPUT;        // float clock high   
        delay_10us();   
    }   
    i2c_stop();   
    INTERRUPT_GLOBAL_ENABLE();   
}

/** \brief High level read from internal EEPROM.  
  
  This function is the user interface to read data from the

UBYTE adc_continue( void ) {   
    if ( ++adc.ui8_Index < ADC_SIGNALS )   
        adc_start( ADC_AUTO );   
    else   
        adc.ui8_Index = 0;   
    return adc.ui8_Index;   
}   
   
/** \brief High level read from ADC (interrupt version).  
  
  This function is the user interface to read data from the  
  internal Analog-to-Digital Converter when using ADC interrupts.  
  \return A 16-bit unsigned integer containing the 10-bit or 8-bit ADC reading.  
  \sa adc_start(), adc_read().  
  \ingroup adc_functions  
*/   
UINT16 adc_get_adres( void ) {   
    UINT16 _adres;   
    if ( ADCON1&0x80 ) {   
        LowByte(_adres) = ADRESL;   
        HighByte(_adres) = ADRESH;   
    }   
    else {   
        // Assumes that by justifying to the left   
        // an ADC resolution of 8-bit is desired   
        _adres = ADRESH;   
    }   
    return _adres;   
}

/** \brief High level read from ADC (polling version).
