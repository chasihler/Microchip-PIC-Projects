; 1  RA2/AN2	: 0-5V Frequency Knob, -24 to +24 semitones
; 2  RA3/AN3	: 0-5V Fine Knob, -100 to +100 cents
; 3  RA4/AN5	: 0-5V Waveform Knob, 0-255, 16 waveforms with interp
; 4  ~MCLR/Vpp	: ICSP Vpp only
; 5  Gnd
; 6  RB0/CCP1	: Main Osc 8-bit PWM Output at 125KHz
; 7  RB1/AN11	: 0-5V Sub Select Knob, 8 waveforms over 4 octaves
; 8  RB2/AN10	: 0-5V Glide Knob, portamento time
; 9  RB3/AN9	: 0-5V Bitcrush Knob, gives 8-bit to 1-bit settings
; 10 RB4/AN8	: 0-5V Note CV, MIDI notes 0-63
; 11 RB5/AN7	: 0-5V Freq Mod CV, -8 to +8 semitones
; 12 ICSPCLK
; 13 ICSPDAT/Glide In	: +5V=Glide On, 0V=Glide Off
; 14 +5V
; 15 RA6		: Currently test output
; 16 RA7		: Sub Osc 8-bit PWM Output at 125KHz
; 17 RA0/AN0	: 0-5V Waveform CV, scans wavetable
; 18 RA1/AN1	: 0-5V Sub Select CV, modulates sub osc selection

; Developed from VCDO1M.ASM, the last/latest test version.

	LIST R=DEC
	INCLUDE <p16f1847.inc>
	Errorlevel -302, -303

; 16F1847 Configuration
	__CONFIG _CONFIG1, _FOSC_INTOSC & _WDTE_OFF & _PWRTE_ON & _MCLRE_OFF & _CP_OFF & _CPD_OFF & _BOREN_OFF & _CLKOUTEN_OFF & _IESO_OFF & _FCMEN_OFF
	__CONFIG _CONFIG2, _WRT_OFF & _PLLEN_ON & _STVREN_ON & _LVP_OFF

;----------------------------------------------------------------------------
; Four sample buffers of 16 samples each
;----------------------------------------------------------------------------
 CBLOCK 0x0A0
	MAIN_OSC_BUFF1: 16			; This needs to be on a multiple of 16 
	SUB_OSC_BUFF1: 16
	MAIN_OSC_BUFF2: 16			; (0x20,0x30 etc)
	SUB_OSC_BUFF2: 16
 ENDC

;----------------------------------------------------------------------------
;	Variables
;----------------------------------------------------------------------------
 CBLOCK 0x020
	; Where are we in the buffer?
	SAMPLE_POS
	; How many samples have we generated? (out of 16 in the buffer)
	LOOP_COUNT

	; General working storage
	INDEX						; Used to store the index during lookups
	TEMP
	TEMP2

	ADC_CHANNEL					; Which of the knobs and CVs are we reading?
	ADC_STAGE					; What stage of the conversion process is this?
	ADC_VALUE					; What's the final value?
	
	; Various flag bits. 1 is on,obviously.
	SYS_FLAGS					; See Define statements below for details

; The Input Variables (Control knobs and CV inputs)
;--------------------------------------------------
	NOTE_CV						; This is basically a MIDI note number 0-119
	FREQ_MOD_CV					; FM input with no stepping
	SUB_SELECT_CV				; Sub Osc Waveform switching
	WAVE_CV						; Main Osc Wavetable scan

	FREQ_CTRL					; 5 main controls
	FINE_CTRL
	WAVE_CTRL
	SUB_SELECT_CTRL				; SubOsc waveform select
;	GLIDE_CTRL					; (In truth, these variables aren't used)
	BITCRUSH_CTRL

;	PITCH_TUNE_CTRL			; Not implemented
;	UNUSED_CTRL				; Currently ISCP input

; Intermediate variables
;------------------------------
	FREQ_OFFSET					; -24 to +24 semitones
	GLIDE_RATE_HI				; The portamento rate
	GLIDE_RATE_LO
	SUB_DETUNE_HI				; Linear detuning of Sub Osc
	SUB_DETUNE_LO	
	BIT_MASK					; Bitcrush mask
	REQ_PITCH_HI				; This is the pitch we're aiming at
	REQ_PITCH_LO
	PITCH_HI					; This is the current pitch
	PITCH_MID					; It may not be the same as REQ_PITCH
	PITCH_LO					; because of portamento
	OCTAVE						; Current octave (working storage)

; The Oscillator + Sub
;------------------------------
	OSC_FREQ_HI					; The 24-bit frequency increment
	OSC_FREQ_MID
	OSC_FREQ_LO

	OSC_PHASE_HI				; The 24-bit phase accumulator
	OSC_PHASE_MID
	OSC_PHASE_LO

	SUB_FREQ_HI					; Same again for Sub Osc
	SUB_FREQ_MID
	SUB_FREQ_LO

	SUB_PHASE_HI
	SUB_PHASE_MID
	SUB_PHASE_LO

	OSC_WAVE					; Waveform selection 0-7
	OSC_SHAPE					; Waveshape - an interp index between 2 waves

	OSC_SUB_OCTAVE				; Sub Osc octave: +1, 0, -1, -2
	OSC_SUB_WAVE				; Address of Sub Osc waveform

; The Waveshape Interpolation
;------------------------------
	UPPER_WAVE					; Address of upper wavetable
	LOWER_WAVE					; Address of lower wavetable
	INPUT_X						; The two things we're interpolating
	INPUT_Y
;	INTERP_INDEX	- I don't need this. I use OSC_SHAPE directly.
	INTERP_OUT					; The result
; 5 bytes free here!
 ENDC

; 0x70-0x7F  Common RAM - Special variables available in all banks
 CBLOCK 0x070
	OUTPUT						; The 8-bit PWM audio oscillator output
	SUB_OUTPUT					; The 8-bit PWM audio sub oscillator output	
 ENDC

;----------------------------------------------------------------------------
;	DEFINE STATEMENTS
;----------------------------------------------------------------------------

; Useful bit definitions for clarity	
#define ZEROBIT				STATUS,Z		; Zero Flag
#define CARRY				STATUS,C		; Carry
#define BORROW				STATUS,C		; Borrow is the same as Carry
; System Flag bit definitions
#define SAMPLES_REQ			SYS_FLAGS, 0	; New samples are required
#define	LOWER_THAN_REQ		SYS_FLAGS, 1	; Pitch is lower than required
#define GLIDE_ON			SYS_FLAGS, 2	; Is Glide switched on? (1=yes)

; Sub Oscillator flags
#define ONE_OCTAVE_DOWN		OSC_SUB_OCTAVE, 5	; Shift SubOsc 1 oct down
#define TWO_OCTAVES_DOWN	OSC_SUB_OCTAVE, 6	; Shift SubOsc 2 octaves down

; Sneaky sample handling flags (see notes)
#define BUFFER_OVERFLOW		SAMPLE_POS, 4	; Have we run out of buffer?
#define BUFFER_ONE			SAMPLE_POS, 5	; Are we using buffer two?

#define GLIDE_SWITCH		PORTB,7			; Glide On/Off input on RB7

; DEBUG - OUTPUT TIMING PULSE
#define TIMING_LED			PORTA,6			; Output LED on RA6

;----------------------------------------------------------------------------
;	In the beginning, all was 0x000 and NOP...
;----------------------------------------------------------------------------
	org     0x0000				; Processor reset vector
	nop							; for ICD use
	goto    Main				; Go to the main program

;----------------------------------------------------------------------------
;	Sample Output Interrupt
;----------------------------------------------------------------------------
	org     0x0004				;Interrupt vector location
InterruptEnter:
	movlb	D'0'				; Bank 0
	bcf		PIR1, TMR2IF		; Clear TMR2 interrupt flag

; Set up the sample pointer
	clrf	FSR1H
	movf	SAMPLE_POS, w		; Add the current position in the buffer
	movwf	FSR1L				; Set up the indirection register
; Get the sample for the Main Osc
	movf	INDF1, w			; Get the main oscillator sample
	movwf	OUTPUT				; Get it ready for output
; Get the sample for the Sub Osc
	moviw	16[FSR1]			; Get the sub oscillator sample
	movwf	SUB_OUTPUT			; Get it ready for output

; Move onto the next sample and check for the end of the buffer
	incf	SAMPLE_POS, f
	
; Have we run out of samples in this buffer?
	btfss	BUFFER_OVERFLOW
	goto	PWMOutput

ChangeBuffer:
	; Clear the overflow bit
	bcf		BUFFER_OVERFLOW
	; Toggle the buffer bit
	movlw	0x60				; Move to buffer 1 if on buffer 2
	xorwf	SAMPLE_POS, f		; and vice versa
	; Request more samples from the main code
	bsf		SAMPLES_REQ	

; Puts the value of OUTPUT/SUB_OUTPUT into the appropriate
; PWM duty cycle registers.
PWMOutput:
; Main Oscillator output
	; Put the 2 LSBs of OUTPUT into CCP1CON
	movlb	D'5'				; Bank 5
	lsrf	OUTPUT, f			; Rotate bit 0 into the carry bit
	bsf		CCP1CON, DC1B0		; Set or clear bit 4 of the CCP1CON register
	btfss	CARRY
	bcf		CCP1CON, DC1B0
	lsrf    OUTPUT, f			; Rotate bit 1 into the carry bit
	bsf		CCP1CON, DC1B1		; Set or clear bit 5 of the CCP1CON register
	btfss	CARRY
	bcf		CCP1CON, DC1B1
	; Put the 6 high bits into CCPR1L
	movf	OUTPUT, w
	movwf   CCPR1L

; Sub Oscillator output
	; Put the 2 LSBs of SUB_OUTPUT into CCP2CON
	lsrf	SUB_OUTPUT, f		; Rotate bit 0 into the carry bit
	bsf		CCP2CON, DC1B0		; Set or clear bit 4 of the CCP2CON register
	btfss	CARRY
	bcf		CCP2CON, DC1B0
	lsrf    SUB_OUTPUT, f		; Rotate bit 1 into the carry bit
	bsf		CCP2CON, DC1B1		; Set or clear bit 5 of the CCP2CON register
	btfss	CARRY
	bcf		CCP2CON, DC1B1
	; Put the 6 high bits into CCPR2L
	movf	SUB_OUTPUT, w
	movwf   CCPR2L

InterruptExit:
	retfie


;----------------------------------------------------------------------------
;	MAIN PROGRAM
; This sets everything up before moving into the sample gen/ADC reading loop.
;----------------------------------------------------------------------------
Main:
	movlb	D'1'				; Bank 1

	; Set up the clock for 32MHz internal
	movlw	B'11110000'			; 8MHz internal, x4 PLL
	movwf	OSCCON

	; Set up the IO Ports
	movlw	B'00111111'			; All inputs except RA6-Sync, RA7-CCP2 PWM out
	movwf	TRISA
	movlw	B'11111110'			; All inputs except RB0-CCP1 PWM out
	movwf	TRISB

	; Set up the PWM sample interrupt
	bsf		INTCON, GIE			; Enable interrupts
	bsf		INTCON, PEIE		; Enable peripheral interrupts
	bsf		PIE1, TMR2IE		; Enable the PWM interrupt

	; Move CCP1 PWM output to pin RB0 and CCP2 PWM output to pin RA7
	movlb	D'2'				; Bank 2
	movlw	B'00001001'
	movwf	APFCON0
 
	; Set up Timer2 as PWM timebase
	; 32MHz clock is used for Timer2
	movlb	D'0'				; Bank 0
	clrf	TMR2				; Clear Timer2
	movlw	0x3F				; Set up Timer2 period register
	movwf	PR2					; 8-bit, 125KHz PWM resolution
	movlw	B'00001000'			; Postscale /2 (for ISR), Prescale /1, Tmr Off
	movwf	T2CON				; PWM Frequency 125KHz, ISR at 62.5KHz

	movlb	D'5'				; Bank 5
	clrf	CCPTMRS				; All PWM modules use Timer2 for the timebase
	; Set up PWM modules CCP1 & CCP2
	movlw	B'00001100'			; Single output, PWM mode
	movwf	CCP1CON
	movwf	CCP2CON

	; Remember that CCPR1L is the 8 MSBs, and the LSBs are in CCP1CON<5:4>
	movlw	0x20
	movwf	CCPR1L				; 50% duty cycle for starters
	movwf	CCPR2L

	movlb	D'0'				; Bank 0

	; Start PWM up
	bsf		T2CON, TMR2ON		; Turn timer 2 on

; Set up Analog-to-digital convertor and ADC inputs
	movlb	D'3'				; Bank 3
	movlw	B'00011111'			; 5 Analog inputs on RA0-RA4
	movwf	ANSELA
	movlw	B'00111110'			; 5 Analog inputs on RB1-RB5
	movwf	ANSELB
; IGNORING RB6 AND RB7 - CURRENTLY ICSP, WILL BE CVS
	
	movlb	D'1'				; Bank 1
	movlw	B'01100000'			; Left-justified, Fosc/64
	movwf	ADCON1

	; Set up Timer0 as ADC acquisition timer
	movlw	B'00000001'			; Prescale for Timer 0, prescale /4 = 2MHz
	movwf	OPTION_REG


	movlb	D'0'				; Bank 0

	; Set up both indirection pointers for Bank0
	clrf	FSR0H
	clrf	FSR1H

	; Set up initial values of the variables
	clrf	SYS_FLAGS
	clrf	ADC_STAGE			; Start with new channel
	clrf	ADC_CHANNEL			; We start by inc, so Channel 1

	; Set up sample Buffer
	movlw	#MAIN_OSC_BUFF1		; We start on buffer 1
	movwf	SAMPLE_POS
	bsf		SAMPLES_REQ			; And we *need* samples!

	; Set up input variables
	clrf	NOTE_CV
	clrf	FREQ_MOD_CV
	clrf	SUB_SELECT_CV
	clrf	WAVE_CV
	clrf	FREQ_CTRL
	clrf	FINE_CTRL
	clrf	SUB_SELECT_CTRL
	clrf	BITCRUSH_CTRL

	; Set up intermediate variables
	clrf	REQ_PITCH_LO		
	clrf	PITCH_LO
	clrf	PITCH_MID
	movlw	D'69'
	movwf	PITCH_HI			; Concert A, 440Hz
	movwf	REQ_PITCH_HI

	movlw	D'255'				; Default to max increment, no portamento
	movwf	GLIDE_RATE_HI		; The portamento rate
	movwf	GLIDE_RATE_LO

;	movlw	D'2'				; Default to no detune
;	movwf	SUB_DETUNE_HI
;	clrf	SUB_DETUNE_LO

	clrf	OCTAVE

	movlw	D'255'				; Default to no bitcrushing, full 8-bits
	movwf	BIT_MASK

	; Set up oscillator
	clrf	OSC_PHASE_LO		; Clear Osc Accumulators
	clrf	OSC_PHASE_MID
	clrf	OSC_PHASE_HI
	clrf	OSC_FREQ_LO
	clrf	OSC_FREQ_MID
	clrf	OSC_FREQ_HI
	clrf	SUB_PHASE_LO		; Clear Sub Osc Accumulators
	clrf	SUB_PHASE_MID
	clrf	SUB_PHASE_HI
	clrf	SUB_FREQ_LO
	clrf	SUB_FREQ_MID
	clrf	SUB_FREQ_HI
	clrf	OSC_WAVE
	clrf	OSC_SUB_OCTAVE		; Defaults to +1 octave
	clrf	OSC_SUB_WAVE


;----------------------------------------------------------------------------
;	MAIN LOOP
; This checks if further samples are required and generates them if so.
; In the meantime, it checks the ADC channels and sets the control variables.
;----------------------------------------------------------------------------
MainLoop:
	movlb	D'0'				; Bank 0

; Do we need more samples for the audio output?
	btfss	SAMPLES_REQ
	goto	ReadCVs				; Read CVs if no sample required yet

	; DEBUG
	bsf		TIMING_LED

	; Yes, we need more samples
	call	SampleGeneration

	; DEBUG
	movlb	D'0'				; Bank 0
	bcf		TIMING_LED
	
	; That's y'lot!
	goto	MainLoop


;----------------------------------------------------------------------------
;	CONTROL KNOB AND EXTERNAL CV INPUTS
;----------------------------------------------------------------------------
ReadCVs:
; We scan ADC channels sequentially. There are four stages:
; Stage 0	- Increment to new channel and select new ADC channel
; Stage 1	- Wait for acquisition, then start conversion
; Stage 2	- Wait for conversion result, then process result
	movf	ADC_STAGE, w			
	brw							; Relative branch with stage value
SelectADCStage:
	goto	ADCNewChannel
	goto	ADCAcquireAndStart
	goto	ADCWaitAndProcess

ADCNewChannel:
	incf	ADC_CHANNEL, w		; Move to next CV input
	andlw	0xF					; Only need four LSBs
	movwf	ADC_CHANNEL			; Store updated CV input/ADC Channel
	; Select new ADC input
	call	GetADChannel
	movlb	D'1'				; Bank 1
	movwf	ADCON0
	movlb	D'0'				; Bank 0	
	; Now wait for acquisition
	movlw	D'5'				; 250 x 0.4us = 0.1ms acquisistion time
	movwf	TMR0
	bcf		INTCON, T0IF		; Clear overflow flag
	incf	ADC_STAGE,f			; Move to waiting for acquisition stage
	goto	MainLoop

ADCAcquireAndStart:
	; We're waiting - has the ADC settled/acquired the input?
	btfss   INTCON, T0IF		; If Timer0 has timed out, it's settled
	goto	MainLoop			; It's not done yet
	; Ok, it's done, so move to the stage below

ADCStartConversion:
	movlb	D'1'				; Bank 1
	bsf     ADCON0, GO			; Start conversion
	movlb	D'0'				; Bank 0
	incf	ADC_STAGE,f			; Move to waiting for result stage
	goto	MainLoop

ADCWaitAndProcess:
	; We're waiting - has the ADC finished the conversion?
	movlb	D'1'				; Bank 1
	btfsc   ADCON0, GO
	goto	MainLoop			; It's not done yet
	; Ok, it's done, so move to the stage below

ADCProcessResult:
	; Exactly what is this ADC result anyway? Lucky we remember what
	; CV input we got it from!
	movf	ADRESH, w
	movlb	D'0'				; Bank 0
	movwf	ADC_VALUE
	movf	ADC_CHANNEL, w		; Get current channel				
	brw							; Relative branch with channel value
SelectCVInput:
	goto	NoteCV
	goto	FreqModCV
	goto	SubSelectCV
	goto	WaveCV
	goto	FreqControl
	goto	FineControl
	goto	WaveformControl
	goto	SubSelectControl
	goto	NoteCV				; External CV inputs get checked twice
	goto	FreqModCV
	goto	SubSelectCV
	goto	WaveCV
	goto	GlideControl
	goto	BitcrushControl
	goto	DummyCV				; AN5	(Currently ISCP)
	goto	DummyCV				; AN6	(Currently ISCP)

;----------------------------------------------------------------------------
;	EXTERNAL CV INPUTS
; Note that these are inverted to suit external inverting mixers.
; The expected range therefore runs from 0=+5V to 255=0V.
;----------------------------------------------------------------------------
NoteCV:
	comf	ADC_VALUE, w		; Invert ADC value into W
	movwf	NOTE_CV				; Store it
	lsrf	NOTE_CV, f			
	lsrf	NOTE_CV, f			; Reduce to 6-bits, 0-63 MIDI note number
	call	RebuildPitchWord	; This will have altered!
	; All done! Go back to the beginning!
	clrf	ADC_STAGE			; Move to the next CV input
	goto	MainLoop

FreqModCV:
	comf    ADC_VALUE, w		; Deal with the inversion
	movwf	FREQ_MOD_CV
	call	RebuildPitchWord	; This will have altered!
	; All done! Go back to the beginning!
	clrf	ADC_STAGE			; Move to the next CV input
	goto	MainLoop

SubSelectCV:
	comf    ADC_VALUE, w		; Deal with the inversion
	movwf	SUB_SELECT_CV
	call	UpdateSubWaveform
	clrf	ADC_STAGE			; Move to the next CV input
	goto	MainLoop

WaveCV:
	comf    ADC_VALUE, w		; Deal with the inversion
	movwf	WAVE_CV
	call	UpdateWaveform
	clrf	ADC_STAGE			; Move to the next CV input
	goto	MainLoop

;----------------------------------------------------------------------------
;	CONTROL KNOBS
;----------------------------------------------------------------------------
FreqControl:
	movf    ADC_VALUE, w
	call	GetFreqControl		; Map 0-255 to -24 to +24 semitones
	btfsc	WREG, 7				; Apply hysteresis (128 represents a boundary)
	goto	FreqControlExit		; Skip if on a boundary
	movwf	FREQ_CTRL			; Store the result
	call	RebuildPitchWord	; This will have altered!
FreqControlExit:
	; All done! Go back to the beginning!
	clrf	ADC_STAGE			; Move to the next CV input
	goto	MainLoop


FineControl:
	movf    ADC_VALUE, w
	movwf	FINE_CTRL
	call	RebuildPitchWord	; This will have altered!
	; All done! Go back to the beginning!
	clrf	ADC_STAGE			; Move to the next CV input
	goto	MainLoop


WaveformControl:
	;  Read the ADC Value and check it out
	movf    ADC_VALUE, w
	movwf	WAVE_CTRL
	call	UpdateWaveform
	; All done! Go back to the beginning!
	clrf	ADC_STAGE			; Move to the next CV input
	goto	MainLoop


SubSelectControl:
	movf    ADC_VALUE, w
	movwf	SUB_SELECT_CTRL
	call	UpdateSubWaveform
	; All done! Go back to the beginning!
	clrf	ADC_STAGE			; Move to the next CV input
	goto	MainLoop


GlideControl:
	; Create the index (shift the ADC_VALUE up one place)
	movf    ADC_VALUE, w		; Store ADC value as index
	movwf	FSR0L
	clrf	FSR0H
	lslf	FSR0L, f			; Shift up for 16-bit table
	rlf		FSR0H, f
	; Add base address to pointer to read the glide data
	movlw	LOW Portamento
	addwf	FSR0L, f
	movlw	HIGH Portamento
	addwfc	FSR0H, f
	; Get the data
	moviw	FSR0++
	movwf	GLIDE_RATE_HI
	movf	INDF0, w			; Annoyingly, we can't do 'moviw FSR0'
	movwf	GLIDE_RATE_LO
	; Is the Glide switched on?
	bsf		GLIDE_ON			; Assume it's on
	btfsc	GLIDE_RATE_HI, 7
	bcf		GLIDE_ON			; Turn off if required
	btfss	GLIDE_SWITCH		; Test glide input
	bcf		GLIDE_ON			; Turn off if required
	; All done! Go back to the beginning!
	clrf	ADC_STAGE			; Move to the next CV input
	goto	MainLoop


BitcrushControl:
; Useful values run 0-15, but only even values are used (0,2,4 etc)
	movlw	0xFF				; Start off with all 8-bits
	movwf	BIT_MASK
	swapf   ADC_VALUE, w		; Reduce it according to CV
	xorlw	B'00001110'			; Invert the bits we use
	andlw	B'00001110'			; Mask unwanted bits
	movwf	BITCRUSH_CTRL
	lsrf	BITCRUSH_CTRL, w	; Get 3-bit version
	brw
	lslf	BIT_MASK, f			; Value 0: 1-bit
	lslf	BIT_MASK, f			; Value 1: 2-bit
	lslf	BIT_MASK, f			; Value 2: 3-bit
	lslf	BIT_MASK, f			; Value 3: 4-bit
	lslf	BIT_MASK, f			; Value 4: 5-bit
	lslf	BIT_MASK, f			; Value 5: 6-bit
	lslf	BIT_MASK, f			; Value 6: 7-bit
	; Value 7 skips to the instruction below (8-bit, no shifts)
	; All done! Go back to the beginning!
	clrf	ADC_STAGE			; Move to the next CV input
	goto	MainLoop
; Could this routine be done better using brw and a table?
; retlw is a 2-cycle instruction, but I could save something?
;
; Currently there's no hysteresis. Don't know how much that matters here.

DummyCV:
	; Do nothing with this reading
	clrf	ADC_STAGE			; Move to the next CV input
	goto	MainLoop


;----------------------------------------------------------------------------
;	Rebuild the Pitch word from the various CV inputs
;----------------------------------------------------------------------------
; The sixteen bit REQ_PITCH_HI/LO word is based on two CVs and three knobs.
;	NOTE_CV		- MIDI Note CV, 0-127, top bit unused
;	FREQ_MOD_CV	- Freq Modulation CV, +/-8 semitones
;	FREQ_CTRL	- Main Frequency knob, +/-24 semitones
;	FINE_CTRL	- Fine frequency knob, +/-100 cents
;	PITCH_TUNE_CTRL	- Pitch Bend or Tuning knob, +/-8 semitones
; Since these are unsigned with centre zero positions, they introduce an
; offset. The sum of these offsets is cancelled by the master offset.

RebuildPitchWord:
	; Start with the Master Offset
	clrf	REQ_PITCH_LO
;	movlw	(256-37)			; -37 semitones offset
	movlw	D'0'				; DEBUG - no offset for now
	movwf	REQ_PITCH_HI

	; Add the Note CV
	movf	NOTE_CV, w
	addwf	REQ_PITCH_HI, f

	; Add the Freq Mod CV
	swapf	FREQ_MOD_CV, w
	andlw	0xF0				; Get low nibble in upper nibble position
	addwf	REQ_PITCH_LO,f		; Add low nibble to low byte
	swapf	FREQ_MOD_CV, w
	andlw	0x0F				; Get upper nibble in low nibble position
	addwfc	REQ_PITCH_HI,f		; Add upper nibble to high byte

	; Add the Frequency control
	movf	FREQ_CTRL, w
	addwf	REQ_PITCH_HI, f

	; Add the Fine control
	lslf	FINE_CTRL, w	
	addwf	REQ_PITCH_LO,f		; Add 7 lowest bits to low byte
	movlw	D'0'
	btfsc	FINE_CTRL, 7
	movlw	D'1'
	addwfc	REQ_PITCH_HI, f		; Add MSB to high byte

	; Add the Pitch tune control
;	swapf	PITCH_TUNE_CV, w
;	andlw	0xF0				; Get low nibble in upper nibble position
;	addwf	REQ_PITCH_LO,f		; Add low nibble to low byte
;	swapf	PITCH_TUNE_CV, w
;	andlw	0x0F				; Get upper nibble in low nibble position
;	addwfc	REQ_PITCH_HI,f		; Add upper nibble to high byte

	; That's the REQ_PITCH worked out!

	; Is the REQ_PITCH higher or lower than the current PITCH?
	movf	PITCH_MID, w
	subwf	REQ_PITCH_LO, w
	movf	PITCH_HI, w			; Do 16-bit comparision
	subwfb	REQ_PITCH_HI, w		; REQ_PITCH - PITCH
	; So is it bigger?
	bsf		LOWER_THAN_REQ
	btfss	BORROW
	bcf		LOWER_THAN_REQ
	return

;----------------------------------------------------------------------------
;	Update Waveform if required
;----------------------------------------------------------------------------
; The waveform is based on a CV and a knob.
;	WAVE_CV		- External CV, -5 to +5V
;	WAVE_CTRL	- Main Waveform knob

; Since WAVE_CV is unsigned with a centre zero position, it introduces an
; offset. Since the sum of the inputs wraps around, it doesn't matter.
UpdateWaveform:
	; What's the sum of the two inputs?
	movf	WAVE_CV, w
	addwf	WAVE_CTRL, w		; It's quite good that these wrap around to 0
	movwf	TEMP
	; Use the upper 4-bits as the WAVE index
	swapf	TEMP, w
	andlw	0x0F
	movwf	OSC_WAVE	
	; Use the lower 4-bits as the SHAPE index
	swapf	TEMP, w
	andlw	0xF0			
	movwf	OSC_SHAPE			; This gets used for between-wave interp

	; Set up waveform addresses for pointers to read waveforms from Prog Mem
	movf	OSC_WAVE, w			; Get the waveform number
	addlw	HIGH Waveforms		; Base address but + 0x8000
	movwf	LOWER_WAVE			; Store base address of lower waveform
	incf	OSC_WAVE, w			; Move to the upper waveform
	andlw	0x0F				; Make sure it wraps around 0-15
	addlw	HIGH Waveforms
	movwf	UPPER_WAVE			; Store base address of upper waveform
	; LOWER_WAVE and UPPER_WAVE are used as FSR1H by the SampleGen code.

	; All done!
	return

;----------------------------------------------------------------------------
;	Update SubOscillator Waveform if required
;----------------------------------------------------------------------------
; The sub osc waveform is based on a CV and a knob.
;	SUB_SELECT_CV	- External CV, -5 to +5V
;	SUB_SELECT_CTRL	- Main Waveform knob

; Since SUB_SELECT_CV is unsigned with a centre zero position, it introduces an
; offset. Since the sum of the inputs wraps around, it doesn't matter.
UpdateSubWaveform:
	; What's the sum of the two inputs?
	movf	SUB_SELECT_CV, w
	addwf	SUB_SELECT_CTRL, w	; It's quite good that these wrap around to 0
	addlw	D'240'				; Reduce offset by 15 (240 = -15 = FUDGE)
	call	GetSubSelectControl	; Map 0-255 to waveform address
	btfss	WREG, 7				; Apply hysteresis (0 represents a boundary)
	goto	SubSelectExit		; Skip if on a boundary
	movwf	TEMP				; Store the result
	andlw	B'01100000'
	movwf	OSC_SUB_OCTAVE		; Store the octave setting
	movf	TEMP, w
	andlw	B'10011111'
	movwf	OSC_SUB_WAVE		; Store the waveform setting	
SubSelectExit:
	; All done!
	return

;----------------------------------------------------------------------------
;	GENERATE A NEW BATCH OF 16 SAMPLES
;----------------------------------------------------------------------------
; There are several stages to this:
; 1) Portamento/Glide -	This looks at our current 24-bit PITCH and
;						moves it towards our 16-bit REQ_PITCH at GLIDE_RATE.
;						Once that is done, we know what the current PITCH is.
; 2) Pitch to FreqInc - We take the current PITCH and convert to FREQ_INC.
; 3) Sample Gen		  -	Actual generation of sixteen samples in a loop.
;						Since this part is repeated 16 times, it must be 
;						highly efficient.

SampleGeneration:
;---------------------------------------------
;	Glide/Portamento
;---------------------------------------------
; Modify the 24-bit PITCH to get closer to the required pitch, REQ_PITCH
; This is done at a constant rate, GLIDE_RATE.

	btfss	GLIDE_ON
	goto	UseReqPitch			; Glide is off, so use REQ_PITCH directly

	; Glide is on - Is it going up or down?
	btfss	LOWER_THAN_REQ
	goto	GlideGoingDown

GlideGoingUp:
	; First, add GLIDE_RATE to the PITCH
	movf	GLIDE_RATE_LO, w	; Add GLIDE_RATE_LO to PITCH_LO
	addwf	PITCH_LO, f
	movf	GLIDE_RATE_HI, w
	addwfc	PITCH_MID, f		; Add to PITCH_MID
	clrw			
	addwfc	PITCH_HI, f			; Add 0 in case of carry to the high byte

	; Second, decide whether that is equal or bigger than the REQ_PITCH
	movf	PITCH_MID, w
	subwf	REQ_PITCH_LO, w
	movf	PITCH_HI, w			; Do 16-bit comparision
	subwfb	REQ_PITCH_HI, w		; REQ_PITCH - PITCH
	; So is it bigger? (Did we overshoot?)
	btfsc	BORROW				; Not negative, we're not there yet
	goto	PitchToFreqInc		; No overshoot, so use this new pitch value
	; It's overshot, so just stop at the required pitch
	goto	UseReqPitch
	; Since we don't alter the LOWER_THAN_REQ flag, it'll do the same
	; process next time, but that doesn't matter.

GlideGoingDown:
	; First, subtract GLIDE_RATE from the PITCH
	movf	GLIDE_RATE_LO, w	; Take GLIDE_RATE_LO from PITCH_LO
	subwf	PITCH_LO, f
	movf	GLIDE_RATE_HI, w
	subwfb	PITCH_MID, f		; Take from PITCH_MID
	clrw
	subwfb	PITCH_HI, f			; Take 0 in case of borrow from high byte

	; Second, decide whether that is equal or less than the REQ_PITCH
	movf	PITCH_MID, w
	subwf	REQ_PITCH_LO, w
	movf	PITCH_HI, w			; Do 16-bit comparision
	subwfb	REQ_PITCH_HI, w		; REQ_PITCH - PITCH
	; So is it less? (Did we undershoot?)
	btfss	BORROW				; Negative, we're not there yet
	goto	PitchToFreqInc		; No undershoot, so use this new pitch value
	; If we get to here, it's undershot, so just stop at the required pitch

UseReqPitch:
	; Just use the required pitch
	movf	REQ_PITCH_HI, w
	movwf	PITCH_HI
	movf	REQ_PITCH_LO, w
	movwf	PITCH_MID
	clrf	PITCH_LO			; IS THIS NECESSARY? (Minor glide time error?)
	
;---------------------------------------------
;	Pitch to Freq Inc conversion
;---------------------------------------------
; Take twelve off the MIDI note number repeatedly until it's 0-11.
; The number of twelves removed is the octave.

PitchToFreqInc:
	movf	PITCH_HI, w			; Get the note number part of the PITCH word

; Octave 0
	clrf	OCTAVE				; Assume Octave 0	
	addlw	D'244'				; 256-12 = 244, so it's the same as -12
	btfss	CARRY				; Less than or equal to 11?
	goto	FoundTheNote		; This must be Octave 0

;Octave 1
	incf	OCTAVE, f
	addlw	D'244'				; Take 12 off it
	btfss	CARRY				; Less than or equal to 11?
	goto	FoundTheNote		; This must be Octave 1

;Octave 2
	incf	OCTAVE, f
	addlw	D'244'				; Take 12 off it
	btfss	CARRY				; Less than or equal to 11?
	goto	FoundTheNote		; This must be Octave 2

;Octave 3
	incf	OCTAVE, f
	addlw	D'244'				; Take 12 off it
	btfss	CARRY				; Less than or equal to 11?
	goto	FoundTheNote		; This must be Octave 3

;Octave 4
	incf	OCTAVE, f
	addlw	D'244'				; Take 12 off it
	btfss	CARRY				; Less than or equal to 11?
	goto	FoundTheNote		; This must be Octave 4

;Octave 5
	incf	OCTAVE, f
	addlw	D'244'				; Take 12 off it
	btfss	CARRY				; Less than or equal to 11?
	goto	FoundTheNote		; This must be Octave 5

;Octave 6
	incf	OCTAVE, f
	addlw	D'244'				; Take 12 off it
	btfss	CARRY				; Less than or equal to 11?
	goto	FoundTheNote		; This must be Octave 6

;Octave 7
	incf	OCTAVE, f
	addlw	D'244'				; Take 12 off it
	btfss	CARRY				; Less than or equal to 11?
	goto	FoundTheNote		; This must be Octave 7

;Octave 8
	incf	OCTAVE, f
	addlw	D'244'				; Take 12 off it
	btfss	CARRY				; Less than or equal to 11?
	goto	FoundTheNote		; This must be Octave 8

;Octave 9
	incf	OCTAVE, f
	addlw	D'244'				; Take 12 off it
	btfss	CARRY				; Less than or equal to 11?
	goto	FoundTheNote		; This must be Octave 9

; We've got a value greater than MIDI note 119
; so let's top it at MIDI note 120
	movlw	0x36				; Set the frequency increment
	movwf	OSC_FREQ_HI			; to 3595800 which is high C,
	movlw	0xDE				; MIDI note 120
	movwf	OSC_FREQ_MID
	movlw	0x18				
	movwf	OSC_FREQ_LO
	goto	Octave9				; No shifts required

; When we jump to here, we've got the correct octave, and *nearly* got the
; note within the octave
FoundTheNote:
; Note number has had one extra set of 12 taken off
	addlw	D'12'				; Range is now 0-11
; Turn this into an index from 0-191
; 11x16 = 176, + up to 15 = 176+15 = 191 max index
	movwf	FSR0L
	swapf	FSR0L, f			; High nibble is 0-11 (0-176)
	swapf	PITCH_MID, w		; Get sub-semitone position
	andlw	0x0F				; I want high nibble in low nibble position
	addwf	FSR0L, f			; Index 0-191 is in FSR0L
	; Now shift the index up one
	clrf	FSR0H
	lslf	FSR0L, f
	rlf		FSR0H, f
	; Set up a pointer to read the Freq Inc data
	movlw	LOW FrequencyIncrement
	addwf	FSR0L, f			; Add base address to pointer
	movlw	HIGH FrequencyIncrement
	addwfc	FSR0H, f
	; Get the data
	moviw	FSR0++
	movwf	OSC_FREQ_HI
	movf	INDF0, w
	movwf	OSC_FREQ_MID
	clrf	OSC_FREQ_LO

; Do two shifts down to get this back into the right place
; (it was upshifted in the lookup tables)
	lsrf	OSC_FREQ_HI, f
	rrf		OSC_FREQ_MID, f
	rrf		OSC_FREQ_LO, f
	lsrf	OSC_FREQ_HI, f
	rrf		OSC_FREQ_MID, f
	rrf		OSC_FREQ_LO, f

; Now, the freq inc needs shifting according to octave.
	movlw	HIGH ShiftByOctave
	movwf	PCLATH
	movf	OCTAVE, w			; Get the octave
	addlw	LOW ShiftByOctave
	btfsc	CARRY
	incf	PCLATH, f
	movwf	PCL
ShiftByOctave:
	goto	Octave0
	goto	Octave1
	goto	Octave2
	goto	Octave3
	goto	Octave4
	goto	Octave5
	goto	Octave6
	goto	Octave7
	goto	Octave8
	goto	Octave9

Octave0:
	lsrf	OSC_FREQ_HI, f
	rrf		OSC_FREQ_MID, f
	rrf		OSC_FREQ_LO, f
Octave1:
	lsrf	OSC_FREQ_HI, f
	rrf		OSC_FREQ_MID, f
	rrf		OSC_FREQ_LO, f
Octave2:
	lsrf	OSC_FREQ_HI, f
	rrf		OSC_FREQ_MID, f
	rrf		OSC_FREQ_LO, f
Octave3:
	lsrf	OSC_FREQ_HI, f
	rrf		OSC_FREQ_MID, f
	rrf		OSC_FREQ_LO, f
Octave4:
	lsrf	OSC_FREQ_HI, f
	rrf		OSC_FREQ_MID, f
	rrf		OSC_FREQ_LO, f
Octave5:
	lsrf	OSC_FREQ_HI, f
	rrf		OSC_FREQ_MID, f
	rrf		OSC_FREQ_LO, f
Octave6:
	lsrf	OSC_FREQ_HI, f
	rrf		OSC_FREQ_MID, f
	rrf		OSC_FREQ_LO, f
Octave7:
	lsrf	OSC_FREQ_HI, f
	rrf		OSC_FREQ_MID, f
	rrf		OSC_FREQ_LO, f
Octave8:
	lsrf	OSC_FREQ_HI, f
	rrf		OSC_FREQ_MID, f
	rrf		OSC_FREQ_LO, f
Octave9:
	; No shift required in octave 9
; We should now have the correct frequency increment

;---------------------------------------------
;	Linear Detune
;---------------------------------------------
; This is intended as an optional replacement for the FINE_CTRL.

	; First add on the DETUNE variable

	; Now remove the 10Hz offset it introduces
;	movlw	D'234'
;	subwf	OSC_FREQ_LO
;	movlw	D'234'
;	subwfb	OSC_FREQ_MID
;	movlw	D'234'
;	subwfb	OSC_FREQ_HI	

;---------------------------------------------
;	Generate Sub Osc Frequency Increment
;---------------------------------------------
	lslf	OSC_FREQ_LO, w		; Start by *up*shifting 1 place
	movwf	SUB_FREQ_LO			; Eg SubOsc can be 1 oct above the main!
	rlf		OSC_FREQ_MID, w
	movwf	SUB_FREQ_MID
	rlf		OSC_FREQ_HI, w
	movwf	SUB_FREQ_HI

; Now downshift the freq inc by the required number of places
ShiftSubDownTwo:
	btfsc	TWO_OCTAVES_DOWN	; Do we shift down two octaves?
	goto	ShiftSubDownOne
	lsrf	SUB_FREQ_HI, f
	rrf		SUB_FREQ_MID, f
	rrf		SUB_FREQ_LO, f	
	lsrf	SUB_FREQ_HI, f
	rrf		SUB_FREQ_MID, f
	rrf		SUB_FREQ_LO, f

ShiftSubDownOne:
	btfsc	ONE_OCTAVE_DOWN		; Do we shift down another octave?
	goto	SubDetune
	lsrf	SUB_FREQ_HI, f
	rrf		SUB_FREQ_MID, f
	rrf		SUB_FREQ_LO, f

SubDetune:
; Detune the suboscillator slightly so it beats against the main osc
	movlw	D'44'				; Beats every 6 seconds
	addwf	SUB_FREQ_LO, f
	clrw
	addwfc	SUB_FREQ_MID, f
	addwfc	SUB_FREQ_HI, f	


;---------------------------------------------
;	Sample Generation Loop
;---------------------------------------------
SetUpSampleLoop:
; Which set of samples are we generating?
; If the interrupt is using buffer1, we generate buffer2 and vice versa
	movlw	#MAIN_OSC_BUFF2		; Assume we build buffer2
	btfss	BUFFER_ONE			; If interrupt is using buffer1, that's ok
	movlw	#MAIN_OSC_BUFF1		; Otherwise, build buffer1
	movwf	FSR0L
	clrf	FSR0H

; Set up a loop that generates a single sample
	movlw	D'16'
	movwf	LOOP_COUNT
OneSampleLoop:

;---------------------------------------------
;	Oscillator
;---------------------------------------------
	; Add 24-bit Freq Inc to 24-bit phase accumulator
	movf	OSC_FREQ_LO, w
	addwf	OSC_PHASE_LO, f
	movf	OSC_FREQ_MID, w
	addwfc	OSC_PHASE_MID, f
	movf	OSC_FREQ_HI, w
	addwfc	OSC_PHASE_HI, f
	
	; Lookup the main Osc waveform
	movf	OSC_PHASE_HI, w		; Use the phase as the low byte of our pointer
;	andwf	BIT_MASK, w			; Apply table derez
	movwf	FSR1L	
	; Read the lower sample
	movf	LOWER_WAVE, w
	movwf	FSR1H
	movf	INDF1, w
	movwf	INPUT_X
	; Read the upper sample
	movf	UPPER_WAVE, w
	movwf	FSR1H
	movf	INDF1, w
	movwf	INPUT_Y

OscInterpolation:
	; Interpolate between them
	clrf	INTERP_OUT

	movf	INPUT_X, w			; 4-bit interp for starters
	btfsc	OSC_SHAPE, 4
	movf	INPUT_Y, w
	addwf	INTERP_OUT, f
	lsrf	INTERP_OUT, f		; Using LSRF saves clearing the carry first

	movf	INPUT_X, w
	btfsc	OSC_SHAPE, 5
	movf	INPUT_Y, w
	addwf	INTERP_OUT, f
	rrf		INTERP_OUT, f

	movf	INPUT_X, w
	btfsc	OSC_SHAPE, 6
	movf	INPUT_Y, w
	addwf	INTERP_OUT, f
	rrf		INTERP_OUT, f

	movf	INPUT_X, w
	btfsc	OSC_SHAPE, 7
	movf	INPUT_Y, w
	addwf	INTERP_OUT, f
	rrf		INTERP_OUT, f
	
	; We've generated a new sample
	movf	INTERP_OUT, w
	andwf	BIT_MASK, w			; Apply bitcrushing
	movwf	INDF0				; Store the waveform output in the buffer

;---------------------------------------------
;	Sub Oscillator
;---------------------------------------------

	; Add 24-bit Freq Inc to 24-bit phase accumulator
	movf	SUB_FREQ_LO, w
	addwf	SUB_PHASE_LO, f
	movf	SUB_FREQ_MID, w
	addwfc	SUB_PHASE_MID, f
	movf	SUB_FREQ_HI, w
	addwfc	SUB_PHASE_HI, f

	; Lookup the Sub Osc waveform
	movf	SUB_PHASE_HI, w		; Get phase for index
;	andwf	BIT_MASK, w			; Apply table derez
	movwf	FSR1L
	movf	OSC_SUB_WAVE, w		; Get waveform address
	movwf	FSR1H
	movf	INDF1, w
	andwf	BIT_MASK, w			; Apply bitcrushing
	movwi	16[FSR0]			; Store the sample in the Sub Osc buffer

	; Move to the next sample in the buffer
	incf	FSR0L, f

	; Do within-block frequency interpolation
	; Add DELTA to the current frequency increment
	
	; Yadda yadda - 16 or 24 bit addition - about 10 instructions
	; Amounts to another 160 in total


; Do we still need to do another sample?
	decfsz	LOOP_COUNT, f
	goto	OneSampleLoop

	; That's all the samples done
	bcf		SAMPLES_REQ
	return
